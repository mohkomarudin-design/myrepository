-- =========================================================================
-- DATABASE SCHEMA SIONE (Sistem Informasi Log Book Dokumen)
-- RDBMS: MariaDB
-- =========================================================================

-- =========================================================================
-- 1. TABEL MASTER DATA
-- =========================================================================

-- Tabel Master Divisi & Cabang
CREATE TABLE IF NOT EXISTS master_divisi (
    id_divisi INT AUTO_INCREMENT PRIMARY KEY,
    kategori VARCHAR(50) NOT NULL,
    nama_divisi VARCHAR(255) NOT NULL UNIQUE,
    CONSTRAINT chk_kategori CHECK(kategori IN ('Kantor Pusat', 'Cabang'))
);

-- Tabel Master Dokumen (Database Utama)
CREATE TABLE IF NOT EXISTS master_dokumen (
    no_dokumen VARCHAR(100) PRIMARY KEY,
    nama_dokumen VARCHAR(500) NOT NULL,
    tahun INT NOT NULL,
    jenis_dokumen VARCHAR(100) NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Tersedia',
    tempat_penyimpanan TEXT,
    nama_klien TEXT,
    nilai_proyek TEXT,
    file_path TEXT,
    CONSTRAINT chk_status CHECK(status IN ('Tersedia', 'Dipinjam', 'Hilang'))
);

-- Tabel Admin Settings (Password & Konfigurasi)
CREATE TABLE IF NOT EXISTS admin_settings (
    id INT PRIMARY KEY,
    password_hash TEXT NOT NULL,
    reset_token TEXT,
    reset_token_expiry TEXT,
    CONSTRAINT chk_id CHECK(id = 1)
);

-- =========================================================================
-- 2. TABEL TRANSAKSI PEMINJAMAN
-- =========================================================================

-- Header Transaksi Peminjaman
CREATE TABLE IF NOT EXISTS trx_peminjaman (
    id_transaksi VARCHAR(50) PRIMARY KEY,
    tgl_pinjam DATE NOT NULL,
    batas_waktu DATE NOT NULL,
    id_divisi INT NOT NULL,
    nama_peminjam VARCHAR(255) NOT NULL,
    email_peminjam VARCHAR(255),
    file_lampiran TEXT,
    ttd_peminjam TEXT,
    ttd_pic TEXT,
    status_keseluruhan VARCHAR(100) NOT NULL DEFAULT 'Menunggu Persetujuan',
    status_approval VARCHAR(50) NOT NULL DEFAULT 'Menunggu',
    catatan_tolak TEXT,
    FOREIGN KEY (id_divisi) REFERENCES master_divisi(id_divisi)
);

-- Detail Transaksi Peminjaman
CREATE TABLE IF NOT EXISTS detail_peminjaman (
    id_detail INT AUTO_INCREMENT PRIMARY KEY,
    id_transaksi VARCHAR(50) NOT NULL,
    no_dokumen VARCHAR(100) NOT NULL,
    status_dokumen VARCHAR(50) NOT NULL DEFAULT 'Dipinjam',
    tgl_dikembalikan DATE,
    FOREIGN KEY (id_transaksi) REFERENCES trx_peminjaman(id_transaksi) ON DELETE CASCADE,
    FOREIGN KEY (no_dokumen) REFERENCES master_dokumen(no_dokumen)
);

-- =========================================================================
-- 3. TABEL LOG PENGEMBALIAN
-- =========================================================================

CREATE TABLE IF NOT EXISTS log_pengembalian (
    id_pengembalian INT AUTO_INCREMENT PRIMARY KEY,
    id_transaksi VARCHAR(50) NOT NULL,
    tgl_kembali DATE NOT NULL,
    nama_pengembali VARCHAR(255) NOT NULL,
    ttd_pengembali TEXT,
    ttd_pic TEXT,
    status_approval VARCHAR(50) NOT NULL DEFAULT 'Menunggu',
    catatan_tolak TEXT,
    FOREIGN KEY (id_transaksi) REFERENCES trx_peminjaman(id_transaksi)
);

-- =========================================================================
-- 4. TABEL TRANSAKSI PENYERAHAN DOKUMEN BARU
-- =========================================================================

CREATE TABLE IF NOT EXISTS trx_penyerahan (
    id_penyerahan VARCHAR(50) PRIMARY KEY,
    tgl_penyerahan DATE NOT NULL,
    nama_penyerah VARCHAR(255) NOT NULL,
    id_divisi INT NOT NULL,
    ttd_penyerah TEXT,
    ttd_pic TEXT,
    status_approval VARCHAR(50) NOT NULL DEFAULT 'Menunggu',
    catatan_tolak TEXT,
    FOREIGN KEY (id_divisi) REFERENCES master_divisi(id_divisi)
);

-- =========================================================================
-- 5. TABEL DETAIL PENYERAHAN
-- =========================================================================

CREATE TABLE IF NOT EXISTS detail_penyerahan (
    id_detail INT AUTO_INCREMENT PRIMARY KEY,
    id_penyerahan VARCHAR(50) NOT NULL,
    no_dokumen VARCHAR(100) NOT NULL,
    nama_dokumen VARCHAR(500) NOT NULL,
    tahun INT NOT NULL,
    jenis_dokumen VARCHAR(100) NOT NULL,
    nama_klien TEXT,
    nilai_proyek TEXT,
    file_path TEXT,
    FOREIGN KEY (id_penyerahan) REFERENCES trx_penyerahan(id_penyerahan) ON DELETE CASCADE
);
